#include "Fomt_Lib.src"

script 1085 Event_1085 {
    const MESSAGE_0 = "Gray wanted to come, too, \r\nbut Saibara wouldn't \r\nlet him. \BRK";
    const MESSAGE_1 = "You're going to race? \r\nGood luck!\BRK";
    
    if ((Check_If_Single(0x1C) == 0)) {
        Free_Event_Entity(0x1C);
    }
    SetEntityFacing(0x1C, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x188) == 0)) {
        TalkOpen();
        Set_Portrait(0x1F);
        Set_Name_Window(0x1C);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x20);
        Set_Name_Window(0x1C);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1C) == 0)) {
        Give_Friendship_Points(0x1C, 5);
    }
    Free_Event_Entity(0x1C);
    SetEntityFacing(0x1C, "Down");
    exit;
}
