#include "Fomt_Lib.src"

script 1169 Event_1169 {
    const MESSAGE_0 = "Good girl...\r\nI love animals, especially \r\ngentle ones like Cows!\BRK";
    const MESSAGE_1 = "I think your Cow has a \r\ngreat face, \xFF!. \BRK";
    
    if ((Check_If_Single(0xA) == 0)) {
        Free_Event_Entity(0xA);
    }
    SetEntityFacing(0xA, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0x64);
        Set_Name_Window(0xA);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x64);
        Set_Name_Window(0xA);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0xA) == 0)) {
        Give_Friendship_Points(0xA, 5);
    }
    Free_Event_Entity(0xA);
    SetEntityFacing(0xA, "Up");
    exit;
}
