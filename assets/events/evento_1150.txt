// Script 1150
#include "Fomt_Lib.src"

script 1150 Event_1150 {
    const MESSAGE_0 = "I like to watch from here. \BRK";
    const MESSAGE_1 = "Good luck, \xFF!. \BRK";
    
    if ((Check_If_Single(0xD) == 0)) {
        Free_Event_Entity(0xD);
    }
    SetEntityFacing(0xD, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x197) == 0)) {
        TalkOpen();
        Set_Portrait(0x75);
        Set_Name_Window(0xD);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x76);
        Set_Name_Window(0xD);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0xD) == 0)) {
        Give_Friendship_Points(0xD, 5);
    }
    Free_Event_Entity(0xD);
    SetEntityFacing(0xD, "Down");
    exit;
}
