// Script 1284
#include "Fomt_Lib.src"

script 1284 Event_1284 {
    const MESSAGE_0 = "Yaaawn!...I'll sure\r\nsleep well tonight!\BRK";
    const MESSAGE_1 = "I heard you won.\r\nThat's great!\BRK";
    const MESSAGE_2 = "Don't sweat it, \r\n\xFF!. \BRK";
    
    if ((Check_If_Single(7) == 0)) {
        Free_Event_Entity(7);
    }
    SetEntityFacing(7, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0xA3);
        Set_Name_Window(7);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x1B1) == 1)) {
            TalkOpen();
            Set_Portrait(0xA1);
            Set_Name_Window(7);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0xA0);
        Set_Name_Window(7);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(7) == 0)) {
        Give_Friendship_Points(7, 5);
    }
    Free_Event_Entity(7);
    SetEntityFacing(7, "Up");
    exit;
}
