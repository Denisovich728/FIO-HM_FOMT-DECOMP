#include "Fomt_Lib.src"

script 1253 Event_1253 {
    const MESSAGE_0 = "I get so excited about the\r\nraces, I can't sleep the \r\nnight before. \BRK";
    const MESSAGE_1 = "If you're going to race,\r\nyou might as well win!\BRK";
    
    if ((Check_If_Single(0x18) == 0)) {
        Free_Event_Entity(0x18);
    }
    SetEntityFacing(0x18, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AB) == 0)) {
        TalkOpen();
        Set_Portrait(0x54);
        Set_Name_Window(0x18);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x54);
        Set_Name_Window(0x18);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x18) == 0)) {
        Give_Friendship_Points(0x18, 5);
    }
    Free_Event_Entity(0x18);
    SetEntityFacing(0x18, "Left");
    exit;
}
