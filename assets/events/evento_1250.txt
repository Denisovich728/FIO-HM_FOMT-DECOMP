// Script 1250
#include "Fomt_Lib.src"

script 1250 Event_1250 {
    const MESSAGE_0 = "It's a perfect day \r\nfor Horse racing!\BRK";
    const MESSAGE_1 = "Are you racing? \r\nI hope you win! \BRK";
    
    if ((Check_If_Single(1) == 0)) {
        Free_Event_Entity(1);
    }
    SetEntityFacing(1, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AB) == 0)) {
        TalkOpen();
        Set_Portrait(0xF);
        Set_Name_Window(1);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xE);
        Set_Name_Window(1);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(1) == 0)) {
        Give_Friendship_Points(1, 5);
    }
    Free_Event_Entity(1);
    SetEntityFacing(1, "Left");
    exit;
}
