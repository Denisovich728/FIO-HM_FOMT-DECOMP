    if ((Check_If_Single(2) == 0)) {
        Free_Event_Entity(2);
    }
    if ((Has_NPC_Talked_Today(2) == 0)) {
        SetEntityFacing(2, GetOppositeFacing(GetEntityFacing(0)));
        TalkOpen();
        Set_Portrait(6);
        Set_Name_Window(2);
        TalkMessage(0);
        TalkClose();

        Set_Name_Window(2);
