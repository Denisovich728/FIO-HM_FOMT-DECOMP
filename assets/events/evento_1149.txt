script 1149 Event_1149 {
    const MESSAGE_0 = "I look forward to this \r\nevery year. \BRK";
    const MESSAGE_1 = "If you're going to enter, \r\nyou may as well win. \BRK";
    
    if ((Check_If_Single(0x18) == 0)) {
        Free_Event_Entity(0x18);
    }
    SetEntityFacing(0x18, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x197) == 0)) {
        TalkOpen();
        Set_Portrait(0x54);
        Set_Name_Window(0x18);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x53);
        Set_Name_Window(0x18);
        TalkMessage(1);

        Give_Friendship_Points(0x18, 5);
    }
    Free_Event_Entity(0x18);
    SetEntityFacing(0x18, "Left");

}
