// Script 1278
#include "Fomt_Lib.src"

script 1278 Event_1278 {
    const MESSAGE_0 = "It's harder to tell what\r\nSheep are thinking than \r\nCows, don't you think? \BRK";
    const MESSAGE_1 = "Your Sheep had such fine\r\nWool I was sure it would\r\nwin! \BRK";
    const MESSAGE_2 = "Don't worry. \r\nThere's always next year.\BRK";
    
    if ((Check_If_Single(2) == 0)) {
        Free_Event_Entity(2);
    }
    SetEntityFacing(2, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(3);
        Set_Name_Window(2);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x1B1) == 1)) {
            TalkOpen();
            Set_Portrait(1);
            Set_Name_Window(2);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0);
        Set_Name_Window(2);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(2) == 0)) {
        Give_Friendship_Points(2, 5);
    }
    Free_Event_Entity(2);
    SetEntityFacing(2, "Down");
    exit;
}
