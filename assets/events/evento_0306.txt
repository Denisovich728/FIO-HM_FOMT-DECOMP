#include "Fomt_Lib.src"

script 306 Event_0306 {
    if ((GetEntityFacing(0) == 0)) {
        if ((Check_Flag(0x61) == 1)) {
            Set_Flag(0x61, 2);
            Add_Festival_Friendship_Bonus(0xC);
            Refresh_Map_Entities();
        } else {
            if ((Check_Flag(0x62) == 1)) {
                Set_Flag(0x62, 2);
                Add_Festival_Friendship_Bonus(0xC);
                Refresh_Map_Entities();
            } else {
                if ((Check_Flag(0x8B) == 1)) {
                    Set_Flag(0x8B, 2);
                    Add_Festival_Friendship_Bonus(0xC);
                    Add_Festival_Friendship_Bonus(7);
                    Refresh_Map_Entities();
                } else {
                    if ((Check_Flag(0x8D) == 1)) {
                        Set_Flag(0x8D, 2);
                        Add_Festival_Friendship_Bonus(0xC);
                        Add_Festival_Friendship_Bonus(7);
                        Refresh_Map_Entities();
                    }
                }
            }
        }
        Fade_Out(2, 1);
        Warp_Player(5, 0x114, 0xB0);
        SetEntityPosition(0, 0x114, 0xB0, 0);
        PlaySong(0, 0xCD);
        Fade_In(2, 1);
    }
    exit;
}
