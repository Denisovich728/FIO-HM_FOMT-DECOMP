                Set_Portrait(0x1A);
                TalkMessage(51);
                break;
            case 0x22:
                if ((Has_NPC_Talked_Today(0x17) == 0)) {
                    Set_Portrait(0x18);
                    TalkMessage(52);
                    goto LBL_0122;
                }
                Set_Portrait(0x17);
                TalkMessage(53);
                break;
            case 0x15:
                if ((Has_NPC_Talked_Today(0x17) == 0)) {
                    Set_Portrait(0x17);
                    TalkMessage(54);
                    goto LBL_0127;
                }
                Set_Portrait(0x18);
                TalkMessage(55);
                break;
        }
        Free_Event_Entity(0x17);
    }
    TalkClose();
    SetEntityFacing(0x17, var_3);
    exit;
