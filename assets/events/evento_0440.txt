// Script 440
#include "Fomt_Lib.src"

script 440 Event_0440 {
    if ((GetEntityFacing(0) == 0)) {
        if ((Check_Flag(0xE9) == 1)) {
            Set_Flag(0xE9, 2);
            Add_Festival_Friendship_Bonus(0x1A);
            Refresh_Map_Entities();
        } else {
            if ((Check_Flag(0xF1) == 1)) {
                Set_Flag(0xF1, 2);
                Add_Festival_Friendship_Bonus(0x1A);
                Refresh_Map_Entities();
                Free_Event_Entity(0x1A);
                Give_Friendship_Points(0x1A, 0x14);
                Free_Event_Entity(7);
                Give_Friendship_Points(7, 0x14);
            }
        }
        Fade_Out(2, 1);
        Warp_Player(1, 0x54, 0x78);
        SetEntityPosition(0, 0x54, 0x78, 0);
        PlaySong(0, 0xCD);
        Fade_In(2, 1);
    }
    exit;
}
