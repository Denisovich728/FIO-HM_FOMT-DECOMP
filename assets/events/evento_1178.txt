// Script 1178
#include "Fomt_Lib.src"

script 1178 Event_1178 {
    const MESSAGE_0 = "I thought that Cow would \r\nwin. It had the best teeth!\BRK";
    const MESSAGE_1 = "Your Cow had good teeth, \r\ntoo. I thought it could win.\BRK";
    const MESSAGE_2 = "It was too bad this year,\r\nbut you should try again \r\nnext year. \BRK";
    
    if ((Check_If_Single(2) == 0)) {
        Free_Event_Entity(2);
    }
    SetEntityFacing(2, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0);
        Set_Name_Window(2);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x19D) == 1)) {
            TalkOpen();
            Set_Portrait(1);
            Set_Name_Window(2);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0);
        Set_Name_Window(2);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(2) == 0)) {
        Give_Friendship_Points(2, 5);
    }
    Free_Event_Entity(2);
    SetEntityFacing(2, "Down");
    exit;
}
