// Script 1092
#include "Fomt_Lib.src"

script 1092 Event_1092 {
    const MESSAGE_0 = "Have you placed a bet yet?\r\nYou do it with the Mayor. \BRK";
    const MESSAGE_1 = "I'm going to bet on you, \r\nso do a good job! \BRK";
    
    if ((Check_If_Single(8) == 0)) {
        Free_Event_Entity(8);
    }
    SetEntityFacing(8, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x188) == 0)) {
        TalkOpen();
        Set_Portrait(0x85);
        Set_Name_Window(8);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x87);
        Set_Name_Window(8);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(8) == 0)) {
        Give_Friendship_Points(8, 5);
    }
    Free_Event_Entity(8);
    SetEntityFacing(8, "Left");
    exit;
}
