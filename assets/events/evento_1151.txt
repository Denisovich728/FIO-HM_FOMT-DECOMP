script 1151 Event_1151 {
    const MESSAGE_0 = "Hi there, \xFF!. \r\nIt's an elimination \r\ntournament, you know. \BRK";
    const MESSAGE_1 = "Good luck. \BRK";
    
    if ((Check_If_Single(8) == 0)) {
        Free_Event_Entity(8);
    }
    SetEntityFacing(8, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x197) == 0)) {
        TalkOpen();
        Set_Portrait(0x85);
        Set_Name_Window(8);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x86);
        Set_Name_Window(8);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(8) == 0)) {
        Give_Friendship_Points(8, 5);
    }
    Free_Event_Entity(8);
    SetEntityFacing(8, "Left");
    exit;
}
