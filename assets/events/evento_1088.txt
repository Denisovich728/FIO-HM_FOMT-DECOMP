// Script 1088
#include "Fomt_Lib.src"

script 1088 Event_1088 {
    const MESSAGE_0 = "What great weather \r\nfor a Horse race!\BRK";
    const MESSAGE_1 = "Good luck! \BRK";
    
    if ((Check_If_Single(0x12) == 0)) {
        Free_Event_Entity(0x12);
    }
    SetEntityFacing(0x12, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x188) == 0)) {
        TalkOpen();
        Set_Portrait(0x50);
        Set_Name_Window(0x12);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x50);
        Set_Name_Window(0x12);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x12) == 0)) {
        Give_Friendship_Points(0x12, 5);
    }
    Free_Event_Entity(0x12);
    SetEntityFacing(0x12, "Left");
    exit;
}
