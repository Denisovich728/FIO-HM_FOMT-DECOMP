// Script 1287
#include "Fomt_Lib.src"

script 1287 Event_1287 {
    const MESSAGE_0 = "The winning Sheep this year\r\nhad a wonderful coat, and\r\nwas so healthy...\BRK";
    const MESSAGE_1 = "Even I got excited \r\nthis year. \BRK";
    const MESSAGE_2 = "It's too bad for you.\r\nDon't get upset, though. \r\nJust try harder next year!\BRK";
    
    if ((Check_If_Single(4) == 0)) {
        Free_Event_Entity(4);
    }
    SetEntityFacing(4, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0xB4);
        Set_Name_Window(4);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x1B1) == 1)) {
            TalkOpen();
            Set_Portrait(0xB5);
            Set_Name_Window(4);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0xB4);
        Set_Name_Window(4);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(4) == 0)) {
        Give_Friendship_Points(4, 5);
    }
    Free_Event_Entity(4);
    SetEntityFacing(4, "Down");
    exit;
}
