// Script 1285
#include "Fomt_Lib.src"

script 1285 Event_1285 {
    const MESSAGE_0 = "I felt warm just \r\nlooking at those Sheep!\BRK";
    const MESSAGE_1 = "Did you win? Would you save\r\nsome of your Wool for me?\r\nIt looks so warm!\BRK";
    const MESSAGE_2 = "Your Sheep lost?\r\nTry not to get too upset. \BRK";
    
    if ((Check_If_Single(0x1B) == 0)) {
        Free_Event_Entity(0x1B);
    }
    SetEntityFacing(0x1B, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0xAD);
        Set_Name_Window(0x1B);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x1B1) == 1)) {
            TalkOpen();
            Set_Portrait(0xAC);
            Set_Name_Window(0x1B);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0xAC);
        Set_Name_Window(0x1B);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1B) == 0)) {
        Give_Friendship_Points(0x1B, 5);
    }
    Free_Event_Entity(0x1B);
    SetEntityFacing(0x1B, "Down");
    exit;
}
