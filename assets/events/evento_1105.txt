// Script 1105
#include "Fomt_Lib.src"

script 1105 Event_1105 {
    const MESSAGE_0 = "The theme for this year is\r\nJuice! I can't wait to \r\ntaste everybody's efforts! \BRK";
    const MESSAGE_1 = "The theme for this year is \r\nDesserts! Make some \r\nirresistible Desserts! \BRK";
    const MESSAGE_2 = "The theme for this year is\r\nBread! Make some great \r\nfresh Bread! \BRK";
    const MESSAGE_3 = "The theme for this year is \r\nNoodles! Make some Noodles \r\nno one can resist! \BRK";
    const MESSAGE_4 = "The theme for this year is \r\nRice. Go make some creative\r\nRice dishes for us! \BRK";
    
    if ((Check_If_Single(0x1E) == 0)) {
        Free_Event_Entity(0x1E);
    }
    SetEntityFacing(0x1E, GetOppositeFacing(GetEntityFacing(0)));
    TalkOpen();
    Set_Portrait(0x3B);
    Set_Name_Window(0x1E);
    switch (Check_Flag(0x18D)) {
        case 0:
            TalkMessage(0);
            break;
        case 1:
            TalkMessage(1);
            break;
        case 2:
            TalkMessage(2);
            break;
        case 3:
            TalkMessage(3);
            break;
        case 4:
            TalkMessage(4);
            break;
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1E) == 0)) {
        Give_Friendship_Points(0x1E, 5);
    }
    Free_Event_Entity(0x1E);
    SetEntityFacing(0x1E, "Down");
    exit;
}
