// Script 300
#include "Fomt_Lib.src"

script 300 Event_0300 {
    if ((GetEntityFacing(0) == 0)) {
        if ((Check_Flag(0xFD) == 1)) {
            Set_Flag(0xFD, 2);
            Add_Festival_Friendship_Bonus(0xA);
            Refresh_Map_Entities();
        }
        Fade_Out(2, 1);
        Warp_Player(0x1A, 0xD8, 0x90);
        SetEntityPosition(0, 0xD8, 0x90, 0);
        Fade_In(2, 1);
    }
    exit;
}
