// Script 1271
#include "Fomt_Lib.src"

script 1271 Event_1271 {
    const MESSAGE_0 = "These Sheep all look so \r\nwell raised and cared for!\BRK";
    const MESSAGE_1 = "I hope your Sheep wins!\BRK";
    
    if ((Check_If_Single(0xD) == 0)) {
        Free_Event_Entity(0xD);
    }
    SetEntityFacing(0xD, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0x75);
        Set_Name_Window(0xD);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x76);
        Set_Name_Window(0xD);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0xD) == 0)) {
        Give_Friendship_Points(0xD, 5);
    }
    Free_Event_Entity(0xD);
    SetEntityFacing(0xD, "Right");
    exit;
}
