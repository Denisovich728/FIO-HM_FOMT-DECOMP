// Script 1290
#include "Fomt_Lib.src"

script 1290 Event_1290 {
    const MESSAGE_0 = "Today is the Pumpkin \r\nFestival. \BRK\WAIT_CLICKYou have to give treats to\r\nchildren. So, please give\r\nsome to me!\BRK";
    const MESSAGE_1 = "I'm waiting!\BRK";
    const MESSAGE_2 = "Thanks a lot!\r\nBy now. \BRK";
    const MESSAGE_3 = "This isn't a treat!\BRK";
    
    var var_0;
    var var_1;
    
    Set_Flag(0x1B2, 1);
    switch (Compare_Gift()) {
        case 0:
            switch (Search_Food_Gift_Address()) {
                case 0x45:
                case 0x65:
                case 0x68:
                case 0x69:
                case 0x71:
                case 0x73:
                case 0x75:
                case 0x8B:
                case 0x8C:
                case 0x8E:
                case 0x8F:
                case 0x90:
                case 0x94:
                case 0x96:
                case 0x97:
                case 0x98:
                case 0xA7:
                case 0xA9:
                    var_0 = 0x10;
                    break;
            }
            break;
    }
    var_1 = GetEntityFacing(5);
    SetEntityFacing(5, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_If_Single(5) == 0)) {
        Free_Event_Entity(5);
    }
    if ((Has_NPC_Talked_Today(5) == 0)) {
        TalkOpen();
        Set_Name_Window(5);
        Set_Portrait(0xB1);
        TalkMessage(0);
        TalkClose();
        Set_Flag(0x1B2, 0);
    } else {
        TalkOpen();
        Set_Name_Window(5);
        if (Inventory_Check()) {
            Set_Portrait(0xB0);
            TalkMessage(1);
            TalkClose();
            Set_Flag(0x1B2, 0);
            goto LBL_000C;
        }
        if ((var_0 == 0x10)) {
            Give_Friendship_Points(5, 0x14);
            Set_Portrait(0xB1);
            TalkMessage(2);
            TalkClose();
            Drop_Held_Item();
            if ((GetEntityX(5) != 0x154)) {
                SetEntityFacing(5, "Right");
                SetEntityAnim(5, 0x9EB);
                Set_Vector_X(5, 0x154, 1);
                Execute_Movement(5);
                SetEntityAnim(5, 0x9E7);
            }
            SetEntityFacing(5, "Up");
            SetEntityAnim(5, 0x9EB);
            Set_Vector_Y(5, 0, 1);
            Execute_Movement(5);
            SetEntityAnim(5, 0x9E7);
            Hide_Entity(5);
            Set_Flag(0x1B2, 2);
            Finalize_Event();
            goto LBL_000C;
        }
        Set_Portrait(0xB3);
        TalkMessage(3);
        TalkClose();
        Set_Flag(0x1B2, 0);
    }
    SetEntityFacing(5, var_1);
    Free_Event_Entity(5);
    exit;
}
