// Script 1258
#include "Fomt_Lib.src"

script 1258 Event_1258 {
    const MESSAGE_0 = "My greatest joy is \r\nobserving everybody's \r\nHorses. \BRK";
    const MESSAGE_1 = "Look like the races are \r\nbeginning. Best of luck! \BRK";
    
    if ((Check_If_Single(0xE) == 0)) {
        Free_Event_Entity(0xE);
    }
    SetEntityFacing(0xE, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AB) == 0)) {
        TalkOpen();
        Set_Portrait(0x91);
        Set_Name_Window(0xE);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x92);
        Set_Name_Window(0xE);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0xE) == 0)) {
        Give_Friendship_Points(0xE, 5);
    }
    Free_Event_Entity(0xE);
    SetEntityFacing(0xE, "Down");
    exit;
}
