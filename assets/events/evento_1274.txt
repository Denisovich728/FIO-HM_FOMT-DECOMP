// Script 1274
#include "Fomt_Lib.src"

script 1274 Event_1274 {
    const MESSAGE_0 = "This Sheep looks like it\r\nhas a good personality. \r\nI like him. \BRK";
    const MESSAGE_1 = "Your sheep all look\r\nvery happy. \BRK";
    
    if ((Check_If_Single(0x1B) == 0)) {
        Free_Event_Entity(0x1B);
    }
    SetEntityFacing(0x1B, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0xAC);
        Set_Name_Window(0x1B);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xAC);
        Set_Name_Window(0x1B);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1B) == 0)) {
        Give_Friendship_Points(0x1B, 5);
    }
    Free_Event_Entity(0x1B);
    SetEntityFacing(0x1B, "Down");
    exit;
}
