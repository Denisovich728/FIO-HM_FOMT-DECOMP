// Script 1261
#include "Fomt_Lib.src"

script 1261 Event_1261 {
    const MESSAGE_0 = "You must become one with\r\nthe Horse as you race \r\ntoward your goal. \BRK";
    const MESSAGE_1 = "You must believe\r\nin yourself. \BRK";
    
    if ((Check_If_Single(6) == 0)) {
        Free_Event_Entity(6);
    }
    SetEntityFacing(6, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AB) == 0)) {
        TalkOpen();
        Set_Portrait(0xA8);
        Set_Name_Window(6);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xA8);
        Set_Name_Window(6);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(6) == 0)) {
        Give_Friendship_Points(6, 5);
    }
    Free_Event_Entity(6);
    SetEntityFacing(6, "Right");
    exit;
}
