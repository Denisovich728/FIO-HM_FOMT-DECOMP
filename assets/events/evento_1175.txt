script 1175 Event_1175 {

    const MESSAGE_1 = "You're going to compete, \r\nright? I hope you win!\BRK";
    
    if ((Check_If_Single(5) == 0)) {
        Free_Event_Entity(5);
    }
    SetEntityFacing(5, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0xB3);
        Set_Name_Window(5);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xB1);
        Set_Name_Window(5);

    }

        Give_Friendship_Points(5, 5);
    }
    Free_Event_Entity(5);
    SetEntityFacing(5, "Left");
    exit;
}
