#include "Fomt_Lib.src"

script 1187 Event_1187 {
    const MESSAGE_0 = "I like Cows because they're\r\nso gentle. \BRK";
    const MESSAGE_1 = "I'm so happy for you! \BRK";
    const MESSAGE_2 = "I'm sorry...\r\nI thought your Cow was \r\nthe best anyway!\BRK";
    
    if ((Check_If_Single(5) == 0)) {
        Free_Event_Entity(5);
    }
    SetEntityFacing(5, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0xB1);
        Set_Name_Window(5);
        TalkMessage(0);
    } else {
        if ((Check_Flag(0x19D) == 1)) {
            TalkOpen();
            Set_Portrait(0xB1);
            Set_Name_Window(5);
            TalkMessage(1);
            goto LBL_0006;
        }
        TalkOpen();
        Set_Portrait(0xB3);
        Set_Name_Window(5);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(5) == 0)) {
        Give_Friendship_Points(5, 5);
    }
    Free_Event_Entity(5);
    SetEntityFacing(5, "Left");
    exit;
}
