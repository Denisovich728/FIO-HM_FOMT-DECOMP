// Script 1096
#include "Fomt_Lib.src"

script 1096 Event_1096 {
    const MESSAGE_0 = "There are so many \r\npretty Horses. \BRK";
    const MESSAGE_1 = "\xFF\" \r\nis going to run, too, \r\nright? \BRK";
    
    if ((Check_If_Single(5) == 0)) {
        Free_Event_Entity(5);
    }
    SetEntityFacing(5, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x188) == 0)) {
        TalkOpen();
        Set_Portrait(0xB1);
        Set_Name_Window(5);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xB0);
        Set_Name_Window(5);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(5) == 0)) {
        Give_Friendship_Points(5, 5);
    }
    Free_Event_Entity(5);
    SetEntityFacing(5, "Right");
    exit;
}
