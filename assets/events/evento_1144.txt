#include "Fomt_Lib.src"

script 1144 Event_1144 {
    const MESSAGE_0 = "Don't get too upset.\r\nThere's always next year!\BRK";
    const MESSAGE_1 = "Great job! You must have\r\npracticed a lot! I look\r\nforward to seeing you \BRK\r\nnext year, too! \BRK";
    const MESSAGE_2 = "The Tournament is always so\r\nfun to watch. Come out\r\nnext year, too! \BRK";
    
    if ((Check_If_Single(0x1C) == 0)) {
        Free_Event_Entity(0x1C);
    }
    SetEntityFacing(0x1C, GetOppositeFacing(GetEntityFacing(0)));
    TalkOpen();
    if ((Check_Flag(0x193) == 1)) {
        if ((Check_Flag(0x195) == 0)) {
            Set_Portrait(0x1F);
            Set_Name_Window(0x1C);
            TalkMessage(0);
        } else {
            Set_Portrait(0x20);
            Set_Name_Window(0x1C);
            TalkMessage(1);
        }
    } else {
        Set_Portrait(0x20);
        Set_Name_Window(0x1C);
        TalkMessage(2);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1C) == 0)) {
        Give_Friendship_Points(0x1C, 5);
    }
    Free_Event_Entity(0x1C);
    SetEntityFacing(0x1C, "Down");
    exit;
}
