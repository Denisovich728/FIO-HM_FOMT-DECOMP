// Script 1171
#include "Fomt_Lib.src"

script 1171 Event_1171 {
    const MESSAGE_0 = "Which Cow do you like\r\nbest? \BRK";
    const MESSAGE_1 = "I'm glad you entered!\r\nWho do you think will win? \BRK";
    
    if ((Check_If_Single(0xD) == 0)) {
        Free_Event_Entity(0xD);
    }
    SetEntityFacing(0xD, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0x76);
        Set_Name_Window(0xD);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x76);
        Set_Name_Window(0xD);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0xD) == 0)) {
        Give_Friendship_Points(0xD, 5);
    }
    Free_Event_Entity(0xD);
    SetEntityFacing(0xD, "Right");
    exit;
}
