script 1097 Event_1097 {
    const MESSAGE_0 = "I wait for this day \r\nthe whole year! \BRK";
    const MESSAGE_1 = "So you're competing? \r\nJust relax and you'll \r\ndo fine. \BRK";
    
    if ((Check_If_Single(4) == 0)) {
        Free_Event_Entity(4);
    }
    SetEntityFacing(4, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x188) == 0)) {
        TalkOpen();
        Set_Portrait(0xB5);
        Set_Name_Window(4);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xB4);
        Set_Name_Window(4);

    }
    TalkClose();

        Give_Friendship_Points(4, 5);
    }
    Free_Event_Entity(4);
    SetEntityFacing(4, "Right");
    exit;
