#include "Fomt_Lib.src"

script 1173 Event_1173 {
    const MESSAGE_0 = "This Cow is so big and \r\nstrong!!\r\nI bet it would flatten you \r\nif you got stepped on!\BRK";
    const MESSAGE_1 = "Your cow looks like a winner\r\nto me, \xFF!. \BRK";
    
    if ((Check_If_Single(0x10) == 0)) {
        Free_Event_Entity(0x10);
    }
    SetEntityFacing(0x10, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x19B) == 0)) {
        TalkOpen();
        Set_Portrait(0x80);
        Set_Name_Window(0x10);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0x81);
        Set_Name_Window(0x10);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x10) == 0)) {
        Give_Friendship_Points(0x10, 5);
    }
    Free_Event_Entity(0x10);
    SetEntityFacing(0x10, "Up");
    exit;
}
