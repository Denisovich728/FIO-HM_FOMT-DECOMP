// Script 1273
#include "Fomt_Lib.src"

script 1273 Event_1273 {
    const MESSAGE_0 = "Just looking at these \r\nSheep makes me want to \r\nput on a Wool sweater!\BRK";
    const MESSAGE_1 = "I bet your Sheep has a \r\ngood chance. \BRK";
    
    if ((Check_If_Single(7) == 0)) {
        Free_Event_Entity(7);
    }
    SetEntityFacing(7, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AF) == 0)) {
        TalkOpen();
        Set_Portrait(0xA0);
        Set_Name_Window(7);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xA6);
        Set_Name_Window(7);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(7) == 0)) {
        Give_Friendship_Points(7, 5);
    }
    Free_Event_Entity(7);
    SetEntityFacing(7, "Up");
    exit;
}
