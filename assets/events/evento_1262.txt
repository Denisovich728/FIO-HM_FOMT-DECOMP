// Script 1262
#include "Fomt_Lib.src"

script 1262 Event_1262 {
    const MESSAGE_0 = "You must believe\r\nin yourself. \BRK";
    const MESSAGE_1 = "I'm betting on you, \r\nso don't fail me!\BRK";
    
    if ((Check_If_Single(0x1B) == 0)) {
        Free_Event_Entity(0x1B);
    }
    SetEntityFacing(0x1B, GetOppositeFacing(GetEntityFacing(0)));
    if ((Check_Flag(0x1AB) == 0)) {
        TalkOpen();
        Set_Portrait(0xAD);
        Set_Name_Window(0x1B);
        TalkMessage(0);
    } else {
        TalkOpen();
        Set_Portrait(0xAD);
        Set_Name_Window(0x1B);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1B) == 0)) {
        Give_Friendship_Points(0x1B, 5);
    }
    Free_Event_Entity(0x1B);
    SetEntityFacing(0x1B, "Down");
    exit;
}
