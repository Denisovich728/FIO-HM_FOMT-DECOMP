script 1090 Event_1090 {
    const MESSAGE_0 = "Horses from Yodel Farm are\r\nracing, too. \BRK\WAIT_CLICKThis year someone else is \r\nracing, but it used to be \r\nBarely. \BRK";
    const MESSAGE_1 = "Get psyched up! \BRK";
    
    if ((Check_If_Single(0x19) == 0)) {
        Free_Event_Entity(0x19);
    }
    SetEntityFacing(0x19, GetOppositeFacing(GetEntityFacing(0)));
    TalkOpen();
    Set_Name_Window(0x19);
    if (!(((((((Check_Flag(0x4C) == 2) || (Check_Flag(0x55) == 2)) || (Check_Flag(0x67) == 2)) || (Check_Flag(0x6F) == 2)) || (Check_Flag(0x173) == 2)) || (Check_Flag(0x86) == 2)) || (Check_Flag(0x87) == 2))) {
        Set_Hearth_Anim(0x19);
    }
    if ((Check_Flag(0x188) == 0)) {
        Set_Portrait(0x57);
        TalkMessage(0);
    } else {
        Set_Portrait(0x59);
        TalkMessage(1);
    }
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x19) == 0)) {
        Give_Friendship_Points(0x19, 5);
    }
    Free_Event_Entity(0x19);
    SetEntityFacing(0x19, "Down");

}
