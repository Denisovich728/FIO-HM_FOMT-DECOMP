script 1224 Event_1224 {
    const MESSAGE_0 = "I wonder why I feel so \r\ntired...\BRK";
    const MESSAGE_1 = "That was delicious!\r\nI see there are no \r\ningredients left over. \BRK";
    
    if ((Check_If_Single(0xB) == 0)) {
        Free_Event_Entity(0xB);
    }
    SetEntityFacing(0xB, GetOppositeFacing(GetEntityFacing(0)));
    if ((Has_Spoken_To_NPC_Today(0xB) == 0)) {
        Give_Friendship_Points(0xB, 5);
    }
    TalkOpen();
    Set_Name_Window(0xB);
    if ((Check_Flag(0x1A5) == 1)) {
        Set_Portrait(0x71);
        TalkMessage(0);
        TalkClose();
        Free_Event_Entity(0xB);
    } else {
        Set_Portrait(0x6F);
        TalkMessage(1);
        TalkClose();
        Free_Event_Entity(0xB);
    }
    SetEntityFacing(0xB, "Down");
    exit;
}
