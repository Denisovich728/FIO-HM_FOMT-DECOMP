script 1240 Event_1240 {
    const MESSAGE_0 = "It's all the more fun to \r\neat with everybody \r\nwhen you live alone. \BRK";
    
    if ((Check_If_Single(0x1B) == 0)) {
        Free_Event_Entity(0x1B);
    }
    SetEntityFacing(0x1B, GetOppositeFacing(GetEntityFacing(0)));
    TalkOpen();
    Set_Portrait(0xAD);
    Set_Name_Window(0x1B);
    TalkMessage(0);
    TalkClose();
    if ((Has_Spoken_To_NPC_Today(0x1B) == 0)) {
        Give_Friendship_Points(0x1B, 5);
    }
    SetEntityFacing(0x1B, "Left");
    Free_Event_Entity(0x1B);
    exit;
}
